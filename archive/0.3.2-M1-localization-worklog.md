`experiments/` because it is a one-off reading of an owner leg, reproduced here):

```js
'use strict';
var B = '<repo>/js/pt/';
var P = require(B + 'params.js'), S = require(B + 'state.js'), SIM = require(B + 'sim.js'), POOL = require(B + 'pool.js');
P.seed = 1; P.sl.kyr = 50;
P.resFrac = P.resMelt + P.resAir + P.resDep;
SIM.init(); SIM.reset(); SIM.dt = 0.05;
POOL.inventoryComp(S);
for (var f = 0; f < 6000; f++) { SIM.step(); POOL.inventoryComp(S); }
var latent = (S.hLedger - S.ledger) - P.meltLat * (S.invLTot - S.invL0);
console.log(JSON.stringify({ fingerprint: S.hash(), emit: S.tx.emit, latent: latent,
	latentRel: Math.abs(latent) / Math.abs(S.hLedger - S.ledger) }));
// → fingerprint 6271680363413605 (the banked p4-check census), emit 75979,
//   latent 4.351e-9 km2*H, 1.085e-13 relative
```
